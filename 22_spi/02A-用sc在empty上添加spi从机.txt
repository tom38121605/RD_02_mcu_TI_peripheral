
（测试uart设置：  uart0， 2M， pb0 tx， pb1 rx）

1.   進入sc界面  （设置完毕后，保存sc到当前工程）     
  
    点中左边的SPI  -- 点击ADD  --   会自动跳出 SPI_0  (下面name改成SPI_1 )

     name：    SPI_1     
 
    quick profiles 设置：      
       spi profiles ：  custom     （默认）
 
    basic configuration 设置：      
       spi initialization configuration 设置：
            mode select： peripheral         （从机）

           clock configuration 设置：
               clock source：    busclk      //32M   （默认）
               clock divider：  divided by 1               

               //target bit rate (hz):   500 000                    //从机无此选项
               //calc bit rate :            500 000.00               //从机无此选项

               frame  format:      motorola 4-wire     （默认）
               clock polarity:       low      （默认）
               phase:                 data capture on first clock   （默认）
               data frame size:   8     （默认）
               bit order:            msb     （默认）

              spi controller basic configuration 设置：
                  chip select:  CS1

    advanced  configuration 设置： 
        Parity：                                disable  （默认）
        rx fifo threshold level：   rx fifo contains >= 1 entry
        tx fifo threshold level：   tx fifo is not full

        spi peripheral advanced  configuration 设置： 
             Enable Clear Shift Ctr On CS Idle:   勾选                    // 当 CS 片选进入空闲 (无效拉高) 时，自动清零内部移位计数器


    interrupt  configuration 设置： 
        enable interrupts：  receive


    Pinmux ...  设置：
       spi peripheral：  SPI1
       sck：     PB9
       pico：   PB8
       poci：   P7
       cs1：    PB17
 

2. keil 魔法棒中设置：  勾选 “reset and run”


3. 在main.c中添加下面的代码

         #include "ti_msp_dl_config.h"

         #include "stdio.h"
         #include "string.h"

         void SendString(char *str);
         char txBuf[100];

         volatile uint8_t gEchoData = 0;
         volatile uint8_t flg_spi=0;

         void delay_ms(uint32_t ms)
         {
             while(ms--)
             {
                 delay_cycles(CPUCLK_FREQ/1000);
             }
         }

          volatile uint8_t gData;

         int main(void)
         {
             SYSCFG_DL_init();

             NVIC_ClearPendingIRQ(SPI_1_INST_INT_IRQN);
             NVIC_EnableIRQ(SPI_1_INST_INT_IRQN);
             gData = 1;

             while (1) 
             {
                 if(flg_spi==1)
                 {
                    flg_spi=0;
                    sprintf(txBuf, "echo: %d\r\n", gData);
                    SendString(txBuf);  
                 }        
                 
             }
         }

         void SPI_1_INST_IRQHandler(void)
         {
             switch (DL_SPI_getPendingInterrupt(SPI_1_INST)) {
                 case DL_SPI_IIDX_RX:

                     gData = DL_SPI_receiveData8(SPI_1_INST);
                     DL_SPI_transmitData8(SPI_1_INST, gData);

                     flg_spi=1;

                     break;
                 default:
                     break;
             }
         }

         void UART_0_INST_IRQHandler(void)
         {
             switch (DL_UART_Main_getPendingInterrupt(UART_0_INST)) 
             {
                 case DL_UART_MAIN_IIDX_RX:

                     gEchoData = DL_UART_Main_receiveData(UART_0_INST);
                     DL_UART_Main_transmitData(UART_0_INST, gEchoData);
                     break;
                 default:
                     break;
             }
         }

         void SendString(char *str)
         {
             while(*str != '\0')
             {
                 DL_UART_Main_transmitDataBlocking(UART_0_INST, *str++);
             }
         }
