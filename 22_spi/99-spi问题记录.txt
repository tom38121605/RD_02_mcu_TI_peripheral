


1.  问： spi 主机发数据给从机（10ms一次）， 如果从机先上电是可以正常接收，并 uart 打印出来 但如果从机拔掉电源，再插上电源，就一直没有反应
    分析： 根本原因
：SPI 是无时钟同步复位协议。

    解决：  增加超时判断 （可考虑用另一个定时器）， 遇到卡住就复位spi0，重新初始化spi0，重新启动spi0

             while (1) 
             {
                 static uint32_t imaincount =0;      
             
                 if(flg_capture_sync==1)  //flg_spi
                 {
                    flg_spi=0;
                    flg_capture_sync=0;                    
                    
                    sprintf(txBuf, "d: %02x\r\n", isynchdata);
                    iputs1(txBuf);  
                    
                    imaincount =0;
                 } 
                
                if (imaincount++ >= 5001000)  //2s
                {
                     imaincount =0;
                     iputs1("run...\r\n");    


                     //NVIC_DisableIRQ(SPI_0_INST_INT_IRQN);    
                     //DL_SPI_disable(SPI_0_INST);  
          
                     DL_SPI_reset(SPI_0_INST);
                     
                     DL_SPI_enablePower(SPI_0_INST);
                     delay_cycles(POWER_STARTUP_DELAY);    

                     SYSCFG_DL_SPI_0_init();


                     //spi0
                     NVIC_ClearPendingIRQ(SPI_0_INST_INT_IRQN);  
                     NVIC_EnableIRQ(SPI_0_INST_INT_IRQN);
            
                }            
                 
             }
