    advanced  configuration 设置：

             enable internal loopback:  不勾选                          //外部要把 PICO 和 POCI短接起来
 

2. keil 魔法棒中设置：  勾选 “reset and run”


3. 在main.c中添加下面的代码

         #include "ti_msp_dl_config.h"

         #include "stdio.h"
         #include "string.h"

         void SendString(char *str);
         char txBuf[100];

         volatile uint8_t gEchoData = 0;

         volatile uint8_t gRxData, gTxData;
         int main(void)
         {
             SYSCFG_DL_init();

             gTxData = 1;

             while (1) 
             {

                 sprintf(txBuf, "data %d %d\r\n", gTxData, gRxData);
                 SendString(txBuf);  

                 DL_SPI_transmitData8(SPI_0_INST, gTxData);
                 gRxData = DL_SPI_receiveDataBlocking8(SPI_0_INST);


                 //__BKPT(0);
                 
                 sprintf(txBuf, "data %d %d\r\n\r\n", gTxData, gRxData);
                 SendString(txBuf);  

                 gTxData++;
             }
         }

         void UART_0_INST_IRQHandler(void)
         {
             switch (DL_UART_Main_getPendingInterrupt(UART_0_INST)) 
             {
                 case DL_UART_MAIN_IIDX_RX:

                     gEchoData = DL_UART_Main_receiveData(UART_0_INST);
                     DL_UART_Main_transmitData(UART_0_INST, gEchoData);
                     break;
                 default:
                     break;
             }
         }

         void SendString(char *str)
         {
             while(*str != '\0')
             {
                 DL_UART_Main_transmitDataBlocking(UART_0_INST, *str++);
             }
         }
