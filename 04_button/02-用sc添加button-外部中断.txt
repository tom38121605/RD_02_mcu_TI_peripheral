1.   進入sc界面  （设配置 button）
  
    点中左边的GPIO  -- 点击ADD  --   会自动跳出 GPIO_GRP_0  (下面由name修改成BUTTON_0)

     name：    BUTTON_0    
     Port：      any   （默认）

    选中 button_0

    group pins   -- 点击ADD  --   会自动跳出  PIN_1 （因led是PIN_0）   // 第一个button引脚不需要点ADD，早已存在
        name：             PIN_1     
        direction：        input   （默认）
        io structure：   any  （默认）

        digital IOMUX features 设置：
            internal resistor：   pull up resistor          //***重要 
            invert：                 disable   （默认）
            input filter：          three ULPCLK cycles   （默认）

        assigned port：       PORTAB
        assigned pin：        21

    interrupts/events  设置：
       enable intrupts：  勾选
       triggle polarity：  triggle on falling edge

    Pinmux ...  设置：
       PIN_0：   PA21/20      //一般由上面的配置自动生成
  

2. keil 魔法棒中设置：  勾选 “reset and run”


3. 在main.c中添加下面的代码

         #include "ti_msp_dl_config.h"

         int main(void)
         {
             SYSCFG_DL_init();

             while (1) 
             {
                uint32_t AAA;
                
                 //delay_cycles(CPUCLK_FREQ/5);

                 AAA = DL_GPIO_readPins(BUTTON_0_PORT, BUTTON_0_PIN_1_PIN);
                 
                 if(AAA)
                     DL_GPIO_setPins(LED_PORT, LED_PIN_0_PIN);
                 else
                     DL_GPIO_clearPins(LED_PORT, LED_PIN_0_PIN);            
             }
             
         }

         #include "ti_msp_dl_config.h"

         volatile uint8_t flg_key1 =0;


         void delay_ms(uint32_t ms)
         {
             while(ms--)
             {
                 delay_cycles(CPUCLK_FREQ/1000);
             }
         }


         int main(void)
         {
             SYSCFG_DL_init();
             
             NVIC_EnableIRQ(BUTTON_0_INT_IRQN);
             

             while (1) 
             {
                 uint32_t AAA;
                 
                 if(flg_key1==1)
                 {
                     flg_key1=0;
                     
                     AAA = DL_GPIO_readPins(BUTTON_0_PORT, BUTTON_0_PIN_1_PIN);
                     if(!AAA)
                     {
                       delay_ms(10);
                       
                       AAA = DL_GPIO_readPins(BUTTON_0_PORT, BUTTON_0_PIN_1_PIN);
                       if(!AAA)
                           DL_GPIO_togglePins(LED_PORT, LED_PIN_0_PIN);
                      }
                 }
             }
             
         }


         void GROUP1_IRQHandler(void)
         {

             if(DL_Interrupt_getStatusGroup(DL_INTERRUPT_GROUP_1, DL_INTERRUPT_GROUP1_GPIOB))
             {
                 DL_Interrupt_clearGroup(DL_INTERRUPT_GROUP_1, DL_INTERRUPT_GROUP1_GPIOB); //清除中断标志
                 
                 if(!DL_GPIO_readPins(BUTTON_0_PORT, BUTTON_0_PIN_1_PIN))    //PB21
                    flg_key1 = 1;
             }
         }
