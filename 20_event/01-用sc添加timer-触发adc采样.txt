

1.  添加adc，進入sc界面  （设置完毕后，保存sc到当前工程）
  
      点中左边的adc12  -- 点击ADD  --   会自动跳出 ADC12_0

     sample clock config设置：
        ADC Clock source：       SYSOSC      (其他时钟源也应该可以考虑)    //默认32M
        sample clock divider：   divide by 1     //默认32M

    sample mode configuration设置：
       conversion mode：                 single
       conversion starting address：  0

       enable repeat mode:             勾选   
       sampling mode:                     auto     (默认)             //下面的time 0 设置的62.5ns自动stop采样
       tigger source          ：            event   （用定时器事件自动触发）


       adc conversion mem confiurations 设置：
          memory 0 设置：
             name：              0
             input channel：        channel2
             device pin name ：   PA25              //由下面的pinmux选择
             reference voltage ： VDDA    (默认)
             sample period source：  sampling timer0    (默认)

          optional  configuration 设置：
             trigger mode：  valid trigger ......    (一个通道转换结束后，不自动触发下一个通道)


    advanced configuration设置：
       conversion resolution：  12-bits             (默认)
       powerdown mode:         manual        
       desired sample time 0:   62.5 ns        //或64us   

    interrupt configuration设置：
       enable interrupts：     mem0 result loaded interrupt 
       interrupt priority；     default                  (默认)


    Event Configuration 设置：
       Event Subscriber Channel ID  ：              1(1- TIMER_0 =>  ADC12_0)


    Pinmux 设置：
        adc12 peripheral：    ADC0
        adc12 channel 2 pin： PA25/26   




1.   添加timer，進入sc界面  （设置完毕后，保存sc到当前工程）
  
    点中左边的TIMER  -- 点击ADD  --   会自动跳出TIMER_0  

     name：                           TIMER_0      
     selected peripheral：       TIMA0     

     timer profiles：   customer   （默认）

    basic config 设置：
       timer clock source：           BUSCLK                        //32M
       timer clock divider:            divided by 1                   //16M
  
       timer clock precaler:         1

       timer clock frequency:       32MHz             （自动计算而出）
       timer period range:           x - y  ms             （自动计算而出可设置的最小至最大延时）

       timer mode:                        periodic down conting      (默认)
       desired  timered period：   1us 


    interrupt config 设置：
       enable interrupt：               无  


    Event Configuration 设置：
       Event 1 Publisher Channel ID ：              1(1- TIMER_0)
       Event 1 Enable Controller Interrupts ：    zero event   



    Pinmux 设置：
       timer peripheral：             TIMA0   


2. keil 魔法棒中设置：  勾选 “reset and run”


3. 在main.c中添加下面的代码

         #include "ti_msp_dl_config.h"
         #include "stdio.h"
         #include "string.h"


         void SendString(char *str);
         char txBuf[100];


         volatile uint8_t flg_adc_mem0;
         volatile uint16_t iadcval;


         void delay_ms(uint32_t ms)
         {
             while(ms--)
             {
                 delay_cycles(CPUCLK_FREQ/1000);
             }
         }


         int main(void)
         {
             SYSCFG_DL_init();
             
             //NVIC_EnableIRQ(TIMER_0_INST_INT_IRQN);    
             DL_TimerA_startCounter(TIMER_0_INST);
            
            
             NVIC_EnableIRQ(ADC12_0_INST_INT_IRQN);   

             //DL_ADC12_startConversion(ADC12_0_INST);

             while (1) 
             {
                 delay_ms(2); 

                 if (flg_adc_mem0==1) 
                 {
                    flg_adc_mem0 = 0;
                    iadcval = DL_ADC12_getMemResult(ADC12_0_INST, DL_ADC12_MEM_IDX_0);
                    
                    sprintf(txBuf, "adc2 %d\r\n", iadcval);
                    SendString(txBuf);       

                    //DL_ADC12_enableConversions(ADC12_0_INST);           
                    //DL_ADC12_startConversion(ADC12_0_INST);
                    
                 }
             }

         }

         void ADC12_0_INST_IRQHandler(void)
         {
             switch (DL_ADC12_getPendingInterrupt(ADC12_0_INST)) {
                 case DL_ADC12_IIDX_MEM0_RESULT_LOADED:
                     flg_adc_mem0 = 1;
                     break;
                 default:
                     break;
             }
         }

         void SendString(char *str)
         {
             while(*str != '\0')
             {
                 DL_UART_Main_transmitDataBlocking(UART_0_INST, *str++);
             }
         }
